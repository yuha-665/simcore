local CHAT_TOKENS_RESERVE = 10000 -- 기본 상태창 생성 컨텍스트 토큰 상한(기본 1만)
local STATUS_BLOCKS_KEEP_IN_REQUEST = 3 -- 상태창용 모델 리퀘스트에 포함할 상태창 블록 최대 개수

-- ==============================
-- [공용 유틸] 상태창 존재 판별 (엄격)
-- ==============================
local function containsStatusBlock(text)
    if not text then return false end
    return (text:find("^%s*★%s*%[") ~= nil) or
           (text:find("^%s*☆%s*%[") ~= nil) or
           (text:find("\n%s*★%s*%[") ~= nil) or
           (text:find("\n%s*☆%s*%[") ~= nil)
end

-- ==============================
-- [공용 유틸] 메시지에서 상태창 블록만 제거 (리퀘스트 컨텍스트용)
--  - 실제 채팅 데이터는 건드리지 않음
--  - 메시지 내부에 "본문 + \n\n ★[...]" 형태로 붙은 상태창을 잘라냄
-- ==============================
local function stripStatusBlockFromText(text)
    if not text then return "" end

    -- 메시지 시작이 상태창이면 전체 제거(빈 문자열 반환)
    if text:find("^%s*[★☆]%s*%[") then
        return ""
    end

    -- 마지막으로 등장하는 "\n  ★[" 또는 "\n  ☆[" 위치를 찾아 그 앞까지만 남김
    local lastPos = nil
    local idx = 1
    while true do
        local s = text:find("\n%s*[★☆]%s*%[", idx)
        if not s then break end
        lastPos = s
        idx = s + 1
    end

    if lastPos then
        local trimmed = text:sub(1, lastPos - 1)
        trimmed = trimmed:match("^%s*(.-)%s*$") or ""
        return trimmed
    end

    return text
end

-- ==============================
-- [EXP 규칙 강제] ☆ 포맷(stats=1)에서만 EXP needed 공식 적용
-- ==============================
local function expNeededNext(level)
    -- Next Lvl Cost = N * 100, 여기서 N = next level (= current+1)
    return (level + 1) * 100
end

local function enforceExpRule(triggerId, status_text)
    local stats = getChatVar(triggerId, "stats") or "0"
    if stats ~= "1" then
        return status_text -- ☆ 포맷에서만 적용
    end

    -- Level 파싱 (Level: 8 또는 Lv 8 둘 다 대응)
    local lvl =
        status_text:match("Level:%s*(%d+)")
        or status_text:match("Lv%s*(%d+)")

    if not lvl then
        return status_text
    end

    local need = expNeededNext(tonumber(lvl))

    -- "EXP: cur / need"에서 need만 교체(첫 1회)
    status_text = status_text:gsub("(EXP:%s*%d+%s*/%s*)%d+", "%1" .. tostring(need), 1)
    return status_text
end

-- 화면 갱신 (변수 변경사항 반영용)
function refreshScreen(triggerId)
    local lastIndex = getChatLength(triggerId) - 1
    if lastIndex >= 0 then
        local lastMsg = getChat(triggerId, lastIndex)
        setChat(triggerId, lastIndex, lastMsg.data)
    end
end

-- 토큰 수 추정
function estimateTokens(text)
    if not text then return 0 end
    return math.ceil(string.len(text) / 3)
end

-- 채팅 기록 포맷팅
-- keepStatusBlocks: nil이면 제한 없음(기존 동작 유지)
-- keepStatusBlocks: 숫자이면, 상태창 블록(★/☆)을 "최신 keepStatusBlocks개"까지만 리퀘스트에 포함
-- tokenLimit: nil이면 CHAT_TOKENS_RESERVE 사용
-- 반환: contextStr, hitLimit, foundStatusInContext
function getContextString(history, keepStatusBlocks, tokenLimit)
    tokenLimit = tokenLimit or CHAT_TOKENS_RESERVE

    local logs = {}
    local current_tokens = 0
    local status_blocks_included = 0
    local hitLimit = false
    local foundStatus = false

    -- 기존과 동일하게 1000 토큰 버퍼 유지
    local budget = tokenLimit - 1000

    for i = #history, 1, -1 do
        local msg = history[i]
        local role = (msg.role == "user") and "User" or "Char"
        local content = msg.data or ""

        if keepStatusBlocks and msg.role == "char" and containsStatusBlock(content) then
            if status_blocks_included < keepStatusBlocks then
                status_blocks_included = status_blocks_included + 1
            else
                -- 3개 초과 상태창은 "리퀘스트 컨텍스트에서만" 제거 (채팅 데이터는 그대로)
                content = stripStatusBlockFromText(content)
            end
        end

        -- 최종적으로 컨텍스트에 포함되는 content 기준으로 "상태창 포함 여부" 체크
        if msg.role == "char" and containsStatusBlock(content) then
            foundStatus = true
        end

        local line = role .. ": " .. content
        local line_tokens = estimateTokens(line)

        if current_tokens + line_tokens > budget then
            hitLimit = true
            break
        end

        current_tokens = current_tokens + line_tokens
        table.insert(logs, 1, line)
    end

    return table.concat(logs, "\n\n"), hitLimit, foundStatus
end

-- 이미 상태창이 있는지 확인 (엄격)
function hasPreviousStatus(history)
    for _, msg in ipairs(history) do
        local content = msg.data or ""
        if msg.role == "char" and containsStatusBlock(content) then
            return true
        end
    end
    return false
end

-- ==============================
-- [추가 유틸] 상태창(★/☆) 블록에서 허용된 헤더만 남기기
-- ==============================
local function _trim(s)
    return (s and (s:match("^%s*(.-)%s*$"))) or ""
end

local function _normalizeKey(k)
    k = _trim(k):lower()
    k = k:gsub("%s+", " ")
    return k
end

local function _buildAllowedKeySet(stats)
    local allowed = {}
    local function add(k) allowed[k] = true end

    -- 공통(★/☆ 모두)
    add("date"); add("time"); add("location"); add("name")
    add("class"); add("weapons"); add("accessories"); add("armor")
    add("items"); add("currencies"); add("skills"); add("quests")

    -- ☆(stats=1)에서만
    if stats == "1" then
        add("level"); add("stat points"); add("exp")
        add("hp"); add("mp"); add("sp")
        add("strength"); add("constitution"); add("agility")
        add("intelligence"); add("sense")
    end

    return allowed
end

local function _extractHeaderKey(seg)
    -- 1) "Key: value" 형태
    local k = seg:match("^%s*([^:|]+)%s*:")
    if k then return k end

    -- 2) "Lv 8", "HP 90/130", "Cur 5000원" 같은 콜론 없는 형태(보험)
    local w = seg:match("^%s*(%w+)%s+")
    if w then return w end

    return nil
end

local function filterExtraStatusHeaders(triggerId, text)
    if not text or text == "" then return text end

    local stats = getChatVar(triggerId, "stats") or "0"
    local allowed = _buildAllowedKeySet(stats)

    local function filterOneBlock(open, inner, close)
        local kept = {}
        local seen = {}

        for seg in inner:gmatch("([^|]+)") do
            seg = _trim(seg)
            if seg ~= "" then
                local rawKey = _extractHeaderKey(seg)
                if rawKey then
                    local canon = _mapToCanonicalKey(rawKey)
                    if canon and allowed[canon] and (not seen[canon]) then
                        seen[canon] = true
                        table.insert(kept, seg)
                    end
                end
            end
        end

        -- 전부 필터되어 빈 블록이 되면, 원문 유지(안전장치)
        -- "정해진 헤더 외에는 무조건 삭제"를 더 엄격히 원하시면 여기만 바꾸시면 됩니다.
        if #kept == 0 then
            return open .. inner .. close
        end

        return open .. table.concat(kept, " | ") .. close
    end

    text = text:gsub("([★☆]%s*%[)(.-)(%])", filterOneBlock)
    return text
end

-- ==============================
-- LLM 출력 텍스트 정제 (triggerId 전달)
-- ==============================
function cleanStatusText(triggerId, text)
    if not text then return "" end

    text = text:match("^%s*(.-)%s*$")
    text = text:gsub("```", "")
    text = text:gsub("<[Tt]houghts?>.-</[Tt]houghts?>", "")
    text = text:gsub("<[Tt]hink>.-</[Tt]hink>", "")
    text = text:gsub("<[Tt]houghts?>", ""):gsub("</[Tt]houghts?>", "")
    text = text:gsub("<[Tt]hink>", ""):gsub("</[Tt]hink>", "")

    local COMMA_GUARD = "__COMMA_GUARD__"

    -- [1] Currencies/Quests 구간의 콤마 보호
    text = text:gsub("(|%s*Currencies:%s*)([^|%]]+)", function(prefix, v)
        return prefix .. v:gsub(",", COMMA_GUARD)
    end)

    text = text:gsub("(|%s*Quests:%s*)([^%]]+)", function(prefix, v)
        return prefix .. v:gsub(",", COMMA_GUARD)
    end)

    -- [2] 숫자 포맷 정리 (1,000 -> 1000)  ※ 보호된 구간은 영향 없음
    while true do
        local new_text, count = text:gsub("(%d+),(%d%d%d)", "%1%2")
        if count == 0 then break end
        text = new_text
    end

    -- [3] 보호 문자 복원 (반드시 숫자 정리 루프 '후'에!)
    text = text:gsub(COMMA_GUARD, ",")

    -- 랭크 표기 정리 (Class 이름 보존 + 랭크만 정리)
    text = text:gsub("(|%s*Class:%s*)([^|%]]+)", function(prefix, content)
        local name, rank = content:match("^(.-)%s*%(([^)]+)%)%s*$")
        if not rank then
            return prefix .. content
        end

        local r_clean = rank:match("^%s*(.-)%s*$")
        local r_lower = r_clean:lower()

        if r_lower == "normal" or r_lower == "hidden" or r_lower == "none" or
           r_clean == "일반" or r_clean == "히든" or r_clean == "없음" then
            return prefix .. name .. " (" .. r_clean .. ") "
        else
            return prefix .. name .. " (none) "
        end
    end)

    -- 스탯 표기 정리
    local patterns = {"Stat Points", "HP", "MP", "SP", "Strength", "Constitution", "Agility", "Intelligence", "Sense"}
    for _, key in ipairs(patterns) do
        text = text:gsub("(|%s*" .. key .. ":%s*[%d%s/]+)%s*%([^%)]*%)", "%1")
    end

    -- 퀘스트 표기 정리
    local s_quest, e_quest = text:find("|%s*Quests:")
    if e_quest then
        local pre = text:sub(1, e_quest)
        local post = text:sub(e_quest + 1)
        if post:match("%]%s*$") then
            local content = post:match("^(.-)%]%s*$")
            if content then
                content = content:gsub("%s*%b[]", "")
                text = pre .. " " .. content .. "]"
            end
        end
    end

    -- ☆ 포맷(stats=1)에서만 EXP needed를 공식으로 강제
    text = enforceExpRule(triggerId, text)

    -- [추가] 정해진 헤더 외 항목 제거
    text = filterExtraStatusHeaders(triggerId, text)

    return text
end

-- ==============================
-- 상태창 생성 실행 (LLM 호출 포함)
-- ==============================
function processStatusWindow(triggerId)
    -- 0: Lua (LLM 호출), 1: Lorebook (작동 중지)
    local status_type = getChatVar(triggerId, "status_type")
    if status_type == "1" then
        return
    end

    local model_mode = getChatVar(triggerId, "status_model") or "2"
    if model_mode == "0" then return end

    -- (중요) LLM 호출 전, 마지막 메시지 상태/역할로 빠른 탈출: 재트리거 비용 누수 방지
    local lastIndex = getChatLength(triggerId) - 1
    if lastIndex < 0 then return end

    local lastMsg = getChat(triggerId, lastIndex)
    if not lastMsg then return end
    if lastMsg.role ~= "char" then return end
    if containsStatusBlock(lastMsg.data or "") then return end

    local history = getFullChat(triggerId)
    if #history == 0 then return end

    local is_first_generation = not hasPreviousStatus(history)
    local user_info_block = ""

    if is_first_generation then
        log("StatusScript: First generation. Injecting profile.")
        local user_name = cbs("{{user}}")
        local user_desc = cbs("{{persona}}")

        user_info_block = "\n\n[Reference: User Profile]\n" ..
                          "Name: " .. user_name .. "\n" ..
                          "Desc: " .. user_desc .. "\n" ..
                          "(Use this to determine initial Rank/Stats if none exist)\n\n"
    end

    -- ==============================
    -- [요구사항 반영]
    -- 1) 기본 10k로 컨텍스트 생성
    -- 2) 10k가 꽉 찼는데(hitLimit) 컨텍스트 안에 상태창이 없으면(foundStatus==false),
    --    13k -> 16k로 늘려서 "컨텍스트 생성만" 재시도
    -- 3) 16k까지 늘려도 없으면, 다시 10k 컨텍스트로 LLM 시도
    -- (LLM 호출은 최종적으로 1회)
    -- ==============================
    local baseLimit = CHAT_TOKENS_RESERVE -- 10000
    local context_str, hit, found = getContextString(history, STATUS_BLOCKS_KEEP_IN_REQUEST, baseLimit)

    if (not is_first_generation) and hit and (not found) then
        local ctx13, hit13, found13 = getContextString(history, STATUS_BLOCKS_KEEP_IN_REQUEST, 13000)
        if found13 then
            context_str = ctx13
            log("StatusScript: Context expanded to 13000 (status found).")
        else
            local ctx16, hit16, found16 = getContextString(history, STATUS_BLOCKS_KEEP_IN_REQUEST, 16000)
            if found16 then
                context_str = ctx16
                log("StatusScript: Context expanded to 16000 (status found).")
            else
                -- 요구사항대로: 16k까지 늘렸는데도 상태창이 없으면 10k로 시도(회귀)
                log("StatusScript: No status found up to 16000; fallback to 10000.")
            end
        end
    end

    local system_prompt = makeStatusPrompt(triggerId, is_first_generation)
    local final_prompt =
        system_prompt .. user_info_block ..
        "--- CHAT LOG ---\n" .. context_str .. "\n--- END LOG ---\nGenerate status."

    local request_msgs = { { role = "user", content = final_prompt } }
    log("StatusScript: Requesting LLM (Mode: " .. model_mode .. ")...")

    local result
    if model_mode == "1" then
        result = LLM(triggerId, request_msgs)
    else
        if _G.axLLM then
            result = axLLM(triggerId, request_msgs)
        else
            result = LLM(triggerId, request_msgs)
        end
    end

    if type(result) == "userdata" or (type(result) == "table" and result.await) then
        result = result:await()
    end

    if result and result.success then
        local status_text = result.message
        if not status_text or status_text == "" then
            status_text = result.result
        end

        status_text = cleanStatusText(triggerId, status_text)
        if not status_text or status_text == "" then return end

        local fresh_history = getFullChat(triggerId)
        local fresh_last_idx = #fresh_history
        if fresh_last_idx == 0 then return end

        local fresh_last_msg = fresh_history[fresh_last_idx]
        if fresh_last_msg.role ~= "char" then return end

        -- 중복 생성 방지 (엄격)
        local current_content = fresh_last_msg.data or ""
        if containsStatusBlock(current_content) then return end

        local final_content = current_content .. "\n\n" .. status_text
        setChat(triggerId, fresh_last_idx - 1, final_content)
        log("StatusScript: Complete!")
    else
        local err_msg = (result and result.message) or "Unknown error"
        log("StatusScript: LLM Failed - " .. err_msg)
    end
end

-- ==============================
-- [4] 메인 실행 트리거
-- ==============================
onOutput = async(function(triggerId)
    -- 1. 캐릭터 감지 로직 실행 (에러 무시)
    local status1, err1 = pcall(processCharacterTokens, triggerId)
    if not status1 then log("CharScript ERROR: " .. tostring(err1)) end

    -- 2. 상태창 생성 로직 실행 (에러 무시)
    local status2, err2 = pcall(processStatusWindow, triggerId)
    if not status2 then log("StatusScript ERROR: " .. tostring(err2)) end
end)
